cmake_minimum_required(VERSION 3.28)

set (CMAKE_VERBOSE_MAKEFILE ON)

if(NOT DEFINED RELEASE_DIR)
    set(RELEASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/release")
endif()

project(c-th LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)

if(WIN32)
    set(STLINK_THREAD_SOURCE src/stlink-env/pthreads/stlink_threads_win32.c)
    set(STLINK_MUTEX_SOURCE src/stlink-env/mutex/stlink_mutex_win32.c)
else()
    set(STLINK_THREAD_SOURCE src/stlink-env/pthreads/stlink_threads_posix.c)
    set(STLINK_MUTEX_SOURCE src/stlink-env/mutex/stlink_mutex_posix.c)
endif()

add_library(stlink_env STATIC ${STLINK_THREAD_SOURCE} ${STLINK_MUTEX_SOURCE})
target_include_directories(stlink_env PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/src/stlink-env/pthreads
    ${CMAKE_CURRENT_SOURCE_DIR}/src/stlink-env/atomic
    ${CMAKE_CURRENT_SOURCE_DIR}/src/stlink-env/mutex
)
if(NOT WIN32)
    set(THREADS_PREFER_PTHREAD_FLAG ON)
    find_package(Threads REQUIRED)
    target_link_libraries(stlink_env PUBLIC Threads::Threads)
endif()

add_executable(${PROJECT_NAME} src/main.c)
target_link_libraries(${PROJECT_NAME} PRIVATE stlink_env)

include(CTest)
if(BUILD_TESTING)
    add_test(NAME threading-and-atomics COMMAND ${PROJECT_NAME})
    set_tests_properties(threading-and-atomics PROPERTIES TIMEOUT 30)
endif()


include(GNUInstallDirs)
install(TARGETS ${PROJECT_NAME}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)

get_filename_component(COMPILER_DIR "${CMAKE_C_COMPILER}" DIRECTORY)
set(COMPILER_ENVIRONMENT "")
if(MSVC)
    # Use the same Visual Studio installation, toolset and architecture as CMake.
    # Modern MSVC lives in VC/Tools/MSVC/<version>/bin/Host<host>/<target>.
    if(NOT COMPILER_DIR MATCHES "^(.*)/Tools/MSVC/([^/]+)/bin/Host([^/]+)/([^/]+)$")
        message(FATAL_ERROR "Cannot locate the MSVC environment for ${CMAKE_C_COMPILER}")
    endif()
    set(VCVARSALL "${CMAKE_MATCH_1}/Auxiliary/Build/vcvarsall.bat")
    set(VCVARS_VERSION "${CMAKE_MATCH_2}")
    set(VCVARS_HOST "${CMAKE_MATCH_3}")
    set(VCVARS_TARGET "${CMAKE_MATCH_4}")
    if(NOT EXISTS "${VCVARSALL}")
        message(FATAL_ERROR "MSVC environment script not found: ${VCVARSALL}")
    endif()
    if(VCVARS_HOST STREQUAL VCVARS_TARGET)
        set(VCVARS_ARCH "${VCVARS_TARGET}")
    else()
        set(VCVARS_ARCH "${VCVARS_HOST}_${VCVARS_TARGET}")
    endif()
    file(TO_NATIVE_PATH "${VCVARSALL}" VCVARSALL)
    set(COMPILER_ENVIRONMENT
        "call \"${VCVARSALL}\" ${VCVARS_ARCH} -vcvars_ver=${VCVARS_VERSION}\nif errorlevel 1 goto :error\n")
endif()

# Visual Studio and Ninja Multi-Config need a separate launcher per configuration.
set(LAUNCHER_SUFFIX "")
if(CMAKE_CONFIGURATION_TYPES)
    set(LAUNCHER_SUFFIX "-$<CONFIG>")
endif()
file(GENERATE 
    OUTPUT "${RELEASE_DIR}/$<TARGET_FILE_BASE_NAME:${PROJECT_NAME}>${LAUNCHER_SUFFIX}.bat" 
    CONTENT
    "@echo off
rem This file is generated by CMake. Do not edit.

setlocal EnableExtensions DisableDelayedExpansion

${COMPILER_ENVIRONMENT}
set \"PATH=${COMPILER_DIR};$<TARGET_RUNTIME_DLL_DIRS:${PROJECT_NAME}>;%PATH%\"


\"${CMAKE_COMMAND}\" --build \"${CMAKE_CURRENT_BINARY_DIR}\" --config \"$<CONFIG>\" --target ${PROJECT_NAME}
if errorlevel 1 goto :error
rem Refresh the release copy even when the build is already up to date.
\"${CMAKE_COMMAND}\" -E copy_if_different \"$<TARGET_FILE:${PROJECT_NAME}>\" \"%~dp0$<TARGET_FILE_NAME:${PROJECT_NAME}>\"
if errorlevel 1 goto :error
\"%~dp0$<TARGET_FILE_NAME:${PROJECT_NAME}>\" %*
exit /b %errorlevel%


:error
exit /b %errorlevel%
"
    NEWLINE_STYLE CRLF
)

add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${RELEASE_DIR}"
    COMMAND ${CMAKE_COMMAND} -E echo "Release: ${RELEASE_DIR}/$<TARGET_FILE_NAME:${PROJECT_NAME}>"
    COMMAND ${CMAKE_COMMAND} -E copy
        "$<TARGET_FILE:${PROJECT_NAME}>"
        "${RELEASE_DIR}/$<TARGET_FILE_NAME:${PROJECT_NAME}>"
    VERBATIM
)
